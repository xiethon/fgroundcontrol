cmake_minimum_required(VERSION 3.22)
project(fgc_field_planner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third-party/boost/headers-1.90.0.tar.xz" BOOST_SHA)
if(NOT BOOST_SHA STREQUAL "4b213ed726fa25d25c70df464743bea30bc97788d71822cedc008dd37f59183d")
 message(FATAL_ERROR "Boost archive checksum mismatch")
endif()
set(BOOST_HEADERS "${CMAKE_CURRENT_BINARY_DIR}/boost-headers")
if(NOT EXISTS "${BOOST_HEADERS}/boost/version.hpp")
 file(MAKE_DIRECTORY "${BOOST_HEADERS}")
 file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/third-party/boost/headers-1.90.0.tar.xz" DESTINATION "${BOOST_HEADERS}")
endif()
file(GLOB_RECURSE CORE CONFIGURE_DEPENDS reference/*.cc)
file(GLOB CLIPPER CONFIGURE_DEPENDS third-party/clipper2/Clipper2Lib/src/*.cpp)
add_library(field_core STATIC ${CORE} ${CLIPPER} planning_bridge.cpp)
target_include_directories(field_core PUBLIC reference reference/field reference/field/ccpp reference/field/ccpp/geo reference/field/ccpp/path_planner third-party/clipper2/Clipper2Lib/include "${BOOST_HEADERS}")
if(ANDROID)
 add_library(fgc_field SHARED jni.cpp)
 target_link_libraries(fgc_field PRIVATE field_core)
 target_link_options(fgc_field PRIVATE "-Wl,-z,max-page-size=16384")
else()
 enable_testing()
 add_executable(field_smoke smoke.cpp)
 target_link_libraries(field_smoke PRIVATE field_core)
 add_test(NAME field_regression COMMAND field_smoke)
endif()
